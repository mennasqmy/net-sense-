/* ================= 1) Welcome — particle network ================= */
(function(){
  const canvas = document.getElementById('net');
  const ctx = canvas.getContext('2d');
  let W, H, pts = [], mouse = {x:-1e4, y:-1e4};

  function resize(){
    W = canvas.width  = canvas.offsetWidth;
    H = canvas.height = canvas.offsetHeight;
    const n = Math.min(130, Math.floor(W * H / 11000));
    pts = [];
    for(let i = 0; i < n; i++) pts.push({
      x: Math.random()*W, y: Math.random()*H,
      vx:(Math.random()-.5)*.4, vy:(Math.random()-.5)*.4,
      r: Math.random()*1.7+.7, ph: Math.random()*Math.PI*2,
      light: i % 3 === 0
    });
  }
  window.addEventListener('resize', resize); resize();
  window.addEventListener('mousemove', e=>{
    const r = canvas.getBoundingClientRect();
    mouse.x = e.clientX - r.left; mouse.y = e.clientY - r.top;
  });

  const LINK = 140;
  function frame(t){
    ctx.clearRect(0,0,W,H);
    for(const p of pts){
      const dxm = mouse.x - p.x, dym = mouse.y - p.y, dm2 = dxm*dxm + dym*dym;
      if(dm2 < 240*240){
        p.vx += dxm * .000012; p.vy += dym * .000012;
        const sp = Math.hypot(p.vx, p.vy);
        if(sp > .8){ p.vx = p.vx/sp*.8; p.vy = p.vy/sp*.8; }
      }
      p.x += p.vx; p.y += p.vy;
      if(p.x < 0 || p.x > W) p.vx *= -1;
      if(p.y < 0 || p.y > H) p.vy *= -1;
      const a = .3 + Math.abs(Math.sin(t/1100 + p.ph)) * .4;
      ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, 6.283);
      ctx.fillStyle = (p.light ? 'rgba(124,196,255,' : 'rgba(96,165,250,') + a.toFixed(3) + ')';
      ctx.fill();
    }
    for(let i = 0; i < pts.length; i++){
      for(let j = i+1; j < pts.length; j++){
        const dx = pts[i].x-pts[j].x, dy = pts[i].y-pts[j].y, d2 = dx*dx+dy*dy;
        if(d2 < LINK*LINK){
          const o = (1 - Math.sqrt(d2)/LINK) * .28;
          ctx.strokeStyle = 'rgba(70,140,255,'+o.toFixed(3)+')';
          ctx.lineWidth = 1;
          ctx.beginPath(); ctx.moveTo(pts[i].x,pts[i].y); ctx.lineTo(pts[j].x,pts[j].y); ctx.stroke();
        }
      }
      const p = pts[i], dx = p.x-mouse.x, dy = p.y-mouse.y, d2 = dx*dx+dy*dy;
      if(d2 < 180*180){
        const o = (1 - Math.sqrt(d2)/180) * .5;
        ctx.strokeStyle = 'rgba(124,196,255,'+o.toFixed(3)+')';
        ctx.beginPath(); ctx.moveTo(p.x,p.y); ctx.lineTo(mouse.x,mouse.y); ctx.stroke();
      }
    }
    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
})();

/* ================= 3) Page routing (Welcome → Platform → Explore) ================= */
const welcome = document.getElementById('welcome');
const app     = document.getElementById('app');
const hub     = document.getElementById('hub');
let transitioning = false;
let appScrollY = 0;

function goApp(){
  if(transitioning || welcome.classList.contains('hide')) return;
  transitioning = true;
  welcome.classList.add('hide');
  document.body.classList.remove('locked');
  setTimeout(()=>{
    app.classList.add('show');
    window.scrollTo(0,0);
    runCounters();
    transitioning = false;
  },520);
}

function swapTo(fromEl, toEl, after){
  if(transitioning) return;
  transitioning = true;
  if(fromEl === app) appScrollY = window.scrollY;
  fromEl.classList.add('hide');
  setTimeout(()=>{
    fromEl.classList.remove('show','hide');
    toEl.classList.add('show');
    requestAnimationFrame(()=> window.scrollTo(0, toEl === app ? appScrollY : 0));
    if(after) after();
    transitioning = false;
  },460);
}

function goWelcome(fromEl){
  if(transitioning) return;
  transitioning = true;
  fromEl.classList.add('hide');
  setTimeout(()=>{
    fromEl.classList.remove('show','hide');
    document.body.classList.add('locked');
    welcome.classList.remove('hide');
    window.scrollTo(0,0);
    transitioning = false;
  },460);
}

document.getElementById('enterBtn').addEventListener('click', goApp);
window.addEventListener('keydown', e=>{
  if(e.key === 'Enter' && !welcome.classList.contains('hide')) goApp();
});

document.getElementById('backHome').addEventListener('click', e=>{ e.preventDefault(); goWelcome(app); });
document.getElementById('hubHome').addEventListener('click', e=>{ e.preventDefault(); goWelcome(hub); });

document.getElementById('startBtn').addEventListener('click', ()=> swapTo(app, hub));
document.getElementById('hubBack').addEventListener('click', ()=> swapTo(hub, app));

/* ================= 5) Animated counters ================= */
function runCounters(){
  document.querySelectorAll('[data-count]').forEach(el=>{
    const target = parseFloat(el.dataset.count);
    const dec    = +(el.dataset.dec || 0);
    const suf    = el.dataset.suf || '';
    const dur = 1600, t0 = performance.now();
    (function f(t){
      const p = Math.min(1,(t-t0)/dur);
      const e = 1 - Math.pow(1-p,3);
      const v = target*e;
      el.textContent = (dec ? v.toFixed(dec) : Math.round(v).toLocaleString('en-US')) + suf;
      if(p < 1) requestAnimationFrame(f);
    })(t0);
  });
}
